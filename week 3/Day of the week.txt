import java.time.LocalDate;

class Solution {
    public String dayOfTheWeek(int day, int month, int year) {
        String dayName = LocalDate.of(year, month, day)
                .getDayOfWeek()
                .toString();

        return dayName.charAt(0) + dayName.substring(1).toLowerCase();
    }
}
